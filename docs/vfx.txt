local ReplicatedStorage = game:GetService("ReplicatedStorage")
local RunService = game:GetService("RunService")
local UserInputService = game:GetService("UserInputService")

local Part_Icles = require(ReplicatedStorage:WaitForChild("Part_Icles"))
Part_Icles:Activate()

local VFXEvent = ReplicatedStorage:WaitForChild("VFXEvent")
local VFXPreloadEvent = ReplicatedStorage:WaitForChild("VFXPreloadEvent")

-- =====================================================================
-- CONFIG (mexa so aqui)
-- =====================================================================

-- Orcamento adaptativo por FPS
local FPS_MINIMO = 30            -- abaixo disso, comeca a cortar VFX
local FPS_SAUDAVEL = 50          -- acima disso, libera mais VFX aos poucos
local LIMITE_MINIMO = 6          -- nunca corta pra menos que isso
local LIMITE_MAXIMO = 12         -- teto mesmo em aparelho bom
local JANELA_AMOSTRAS = 30       -- frames na media (suaviza picos)
local INTERVALO_AJUSTE = 0.5     -- segundos entre cada ajuste

local PARTICULAS_MINIMO = 300
local PARTICULAS_MAXIMO = 5000
local PARTICULAS_PASSO = 150

local LIMITE_VFX_MOBILE = 4      -- ponto de partida em celular
local LIMITE_PARTICULAS_MOBILE = 400

-- Culling por distancia (em studs, medido a partir da CAMERA)
local DISTANCIA_MAXIMA = 600     -- alcance em aparelho bom
local DISTANCIA_MINIMA = 250     -- alcance em aparelho sofrendo
local HISTERESE = 1.25           -- VFX ja tocando so desliga a 125% do alcance
local INTERVALO_VARREDURA = 0.5  -- segundos entre varreduras de VFX tocando

-- =====================================================================
-- PRELOAD DE TEXTURA (evita o soluco na primeira vez que um VFX toca)
-- =====================================================================
-- BindableEvent "VFXPreloadEvent" em ReplicatedStorage, 100% client-side:
--   VFXPreloadEvent:Fire(vfxPasta, true)  -- preload
--   VFXPreloadEvent:Fire(vfxPasta, false) -- deload (libera memoria)
-- Chame ANTES do combate (loading, ao equipar arma/habilidade).

VFXPreloadEvent.Event:Connect(function(vfxPasta, preload)
	if typeof(vfxPasta) ~= "Instance" or not vfxPasta.Parent then
		return
	end

	pcall(function()
		if preload then
			Part_Icles:Preload(vfxPasta)
		else
			Part_Icles:Deload(vfxPasta)
		end
	end)
end)

-- =====================================================================
-- LIMITE ADAPTATIVO DE VFX SIMULTANEOS
-- =====================================================================
-- Mede o FPS real e ajusta quantos VFX / particulas podem existir ao
-- mesmo tempo. Funciona em qualquer aparelho, sem tabela de specs.

local UserGameSettings = UserSettings():GetService("UserGameSettings")

local pareceMobile = UserInputService.TouchEnabled and not UserInputService.KeyboardEnabled
local limiteVfxSimultaneos = pareceMobile and LIMITE_VFX_MOBILE or LIMITE_MAXIMO
local limiteParticulas = pareceMobile and LIMITE_PARTICULAS_MOBILE or PARTICULAS_MAXIMO

-- Teto manual: se o jogador baixou a qualidade grafica (fora do
-- "Automatico"), isso vira um maximo que o FPS nunca ultrapassa.
local tetoVfxQualidade = nil
local tetoParticulasQualidade = nil

local function aplicarLimites()
	if tetoVfxQualidade then
		limiteVfxSimultaneos = math.min(limiteVfxSimultaneos, tetoVfxQualidade)
	end
	if tetoParticulasQualidade then
		limiteParticulas = math.min(limiteParticulas, tetoParticulasQualidade)
	end
	Part_Icles.MAX_ACTIVE_PARTICLES = limiteParticulas
end

local function recalcularTetoQualidade()
	local nivelSalvo = UserGameSettings.SavedQualityLevel

	tetoVfxQualidade = nil
	tetoParticulasQualidade = nil

	if nivelSalvo ~= Enum.SavedQualitySetting.Automatic then
		-- Maior nivel calculado dinamicamente, nao quebra se a Roblox mudar o enum.
		local maiorNivel = 0
		for _, item in Enum.SavedQualitySetting:GetEnumItems() do
			if item.Value > maiorNivel then
				maiorNivel = item.Value
			end
		end

		if maiorNivel > 0 then
			local fracao = nivelSalvo.Value / maiorNivel
			tetoVfxQualidade = math.max(LIMITE_MINIMO, math.floor(LIMITE_MAXIMO * fracao))
			tetoParticulasQualidade = math.max(PARTICULAS_MINIMO, math.floor(PARTICULAS_MAXIMO * fracao))
		end
	end

	aplicarLimites()
end

recalcularTetoQualidade()
UserGameSettings:GetPropertyChangedSignal("SavedQualityLevel"):Connect(recalcularTetoQualidade)

-- Janela circular de frames (O(1) por frame, sem table.remove/insert)
local historico = table.create(JANELA_AMOSTRAS, 0)
local indiceHistorico = 0
local framesColetados = 0
local somaFrames = 0
local ultimoAjuste = 0

RunService.Heartbeat:Connect(function(dt)
	if dt <= 0 then
		return
	end

	indiceHistorico = indiceHistorico % JANELA_AMOSTRAS + 1
	somaFrames += dt - historico[indiceHistorico]
	historico[indiceHistorico] = dt
	if framesColetados < JANELA_AMOSTRAS then
		framesColetados += 1
	end

	local agora = os.clock()
	if agora - ultimoAjuste < INTERVALO_AJUSTE then
		return
	end
	ultimoAjuste = agora

	if framesColetados < JANELA_AMOSTRAS then
		return -- ainda juntando amostra pra confiar na media
	end

	local fpsMedio = framesColetados / somaFrames

	if fpsMedio < FPS_MINIMO then
		limiteVfxSimultaneos = math.max(LIMITE_MINIMO, limiteVfxSimultaneos - 1)
		limiteParticulas = math.max(PARTICULAS_MINIMO, limiteParticulas - PARTICULAS_PASSO)
	elseif fpsMedio > FPS_SAUDAVEL then
		limiteVfxSimultaneos = math.min(LIMITE_MAXIMO, limiteVfxSimultaneos + 1)
		limiteParticulas = math.min(PARTICULAS_MAXIMO, limiteParticulas + PARTICULAS_PASSO)
	end

	aplicarLimites()
end)

-- =====================================================================
-- CULLING POR DISTANCIA (nao emite VFX que o jogador nem consegue ver)
-- =====================================================================
-- Pra VFX que devem tocar SEMPRE (ex: Blur/ColorCorrection de tela), ponha
-- o atributo IgnorarCulling = true na pasta.
--
-- O servidor pode mandar a origem junto, e e o jeito mais preciso:
--   VFXEvent:FireAllClients(vfxPasta, hrp.Position)
-- Aceita Vector3, CFrame, BasePart, Attachment ou Model. Sem origem, usa a
-- primeira Part/Attachment encontrada dentro da pasta.

local camera = workspace.CurrentCamera
workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(function()
	camera = workspace.CurrentCamera
end)

-- [vfxPasta] = BasePart/Attachment usado como referencia de posicao
local referencias = setmetatable({}, {__mode = "k"})

local function acharReferencia(vfxPasta)
	if vfxPasta:IsA("BasePart") or vfxPasta:IsA("Attachment") then
		return vfxPasta
	end
	for _, d in vfxPasta:GetDescendants() do
		if d:IsA("BasePart") or d:IsA("Attachment") then
			return d
		end
	end
	return nil
end

local function posicaoDe(vfxPasta, origem)
	local tipo = typeof(origem)
	if tipo == "Vector3" then
		return origem
	elseif tipo == "CFrame" then
		return origem.Position
	elseif tipo == "Instance" then
		if origem:IsA("BasePart") then
			return origem.Position
		elseif origem:IsA("Attachment") then
			return origem.WorldPosition
		elseif origem:IsA("Model") then
			return origem:GetPivot().Position
		end
	end

	local ref = referencias[vfxPasta]
	if not ref or not ref.Parent then
		ref = acharReferencia(vfxPasta)
		referencias[vfxPasta] = ref
	end
	if not ref then
		return nil
	end

	if ref:IsA("Attachment") then
		return ref.WorldPosition
	end
	return ref.Position
end

-- Aparelho sofrendo enxerga menos longe (acompanha o orcamento por FPS).
local function distanciaMaximaAtual()
	local fator = limiteVfxSimultaneos / LIMITE_MAXIMO
	return math.max(DISTANCIA_MINIMA, DISTANCIA_MAXIMA * fator)
end

local function estaPerto(vfxPasta, origem, multiplicador)
	if vfxPasta:GetAttribute("IgnorarCulling") then
		return true
	end
	if not camera then
		return true
	end

	local pos = posicaoDe(vfxPasta, origem)
	if not pos then
		return true -- sem posicao conhecida, na duvida emite
	end

	local maximo = distanciaMaximaAtual() * (multiplicador or 1)
	local delta = pos - camera.CFrame.Position
	return delta:Dot(delta) <= maximo * maximo -- sem raiz quadrada
end

-- =====================================================================
-- SISTEMA DE EMISSAO DE VFX
-- =====================================================================

-- [vfxPasta] = { tocando = bool, emissao = numero da emissao }
local estados = setmetatable({}, {__mode = "k"})
-- Conjunto so dos que estao tocando (a varredura de distancia itera so isso)
local ativos = setmetatable({}, {__mode = "k"})
local emissoesAtivas = 0

local function obterEstado(vfxPasta)
	local estado = estados[vfxPasta]
	if not estado then
		estado = { tocando = false, emissao = 0 }
		estados[vfxPasta] = estado
	end
	return estado
end

-- Unico lugar que muda "tocando": mantem emissoesAtivas e o conjunto
-- "ativos" sempre sincronizados.
local function marcarTocando(vfxPasta, estado, tocando)
	if estado.tocando == tocando then
		return
	end
	estado.tocando = tocando
	ativos[vfxPasta] = tocando or nil
	emissoesAtivas += tocando and 1 or -1
end

local function podeEmitirMais()
	return emissoesAtivas < limiteVfxSimultaneos
end

local function desativarVFX(vfxPasta)
	if not vfxPasta.Parent then
		return
	end
	pcall(Part_Icles.AbsoluteDisable, Part_Icles, vfxPasta)
end

-- Auto-desligamento quando a duracao REAL termina. So agenda se a duracao
-- for finita (AbsoluteEmit retorna nil pra efeitos em loop/infinitos).
local function agendarAutoDesligamento(vfxPasta, estado, idEmissao, duracao)
	if duracao == nil then
		return
	end

	task.delay(duracao, function()
		-- Emissao mais nova ja aconteceu (combo rapido): nao mexe nela.
		if estado.emissao ~= idEmissao then
			return
		end
		if vfxPasta.Parent then
			desativarVFX(vfxPasta)
		end
		marcarTocando(vfxPasta, estado, false)
	end)
end

local function iniciarEmissao(vfxPasta, estado, idEmissao)
	local ok, duracao = pcall(Part_Icles.AbsoluteEmit, Part_Icles, vfxPasta)
	if not ok then
		return
	end

	marcarTocando(vfxPasta, estado, true)
	agendarAutoDesligamento(vfxPasta, estado, idEmissao, duracao)
end

local function emitirVFX(vfxPasta, estado)
	estado.emissao += 1
	local idEmissao = estado.emissao

	-- Combo rapido: ainda estava tocando. Limpa e reemite. A reemissao da
	-- MESMA pasta sempre passa, ela ja tinha lugar no orcamento.
	if estado.tocando then
		marcarTocando(vfxPasta, estado, false)
		desativarVFX(vfxPasta)

		-- Deixa o sistema processar o disable antes de reusar a hierarquia.
		task.defer(function()
			if not vfxPasta.Parent or estado.emissao ~= idEmissao then
				return
			end
			iniciarEmissao(vfxPasta, estado, idEmissao)
		end)
		return
	end

	-- VFX novo: so entra se houver espaco no orcamento. Sobrecarregado,
	-- descarta em silencio em vez de travar.
	if not podeEmitirMais() then
		return
	end

	iniciarEmissao(vfxPasta, estado, idEmissao)
end

VFXEvent.OnClientEvent:Connect(function(vfxPasta, origem)
	if typeof(vfxPasta) ~= "Instance" or not vfxPasta.Parent then
		return
	end

	-- Longe demais: descarta antes de gastar orcamento ou CPU.
	if not estaPerto(vfxPasta, origem) then
		return
	end

	emitirVFX(vfxPasta, obterEstado(vfxPasta))
end)

-- Desliga VFX que ja estavam tocando (loops, stances) e ficaram longe
-- porque o jogador (ou o alvo) se afastou. A histerese evita liga/desliga
-- na borda do alcance.
task.spawn(function()
	while true do
		task.wait(INTERVALO_VARREDURA)

		for vfxPasta in ativos do
			local estado = estados[vfxPasta]
			if not vfxPasta.Parent then
				-- Pasta destruida: so corrige o contador.
				if estado then
					marcarTocando(vfxPasta, estado, false)
				end
			elseif estado and not estaPerto(vfxPasta, nil, HISTERESE) then
				estado.emissao += 1 -- invalida o auto-desligamento pendente
				desativarVFX(vfxPasta)
				marcarTocando(vfxPasta, estado, false)
			end
		end
	end
end)
